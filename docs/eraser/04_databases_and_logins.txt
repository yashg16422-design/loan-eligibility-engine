// Eraser diagram type: CLOUD ARCHITECTURE DIAGRAM
// What it shows: ONE database server, TWO databases inside it, FOUR logins, and which login may touch which tables.
// Source of truth: sql/R__grants.sql (the grants) and loan_engine/migrate/handler.py (creates roles + the n8n database).
// Idea in plain words: every part of the system gets its own key that opens only the doors it needs
// (least privilege). If the API is ever hacked it still cannot write to the matches table.

title Databases and logins - who can open which door

apiLambda [label: "api Lambda - upload API, dashboard, ops view", icon: aws-lambda, color: orange]
ingestLambda [label: "ingest Lambda - reads CSV, loads users", icon: aws-lambda, color: orange]
n8nHost [label: "n8n on EC2 - Workflows A, B, C and error handler", icon: aws-ec2, color: purple]
n8nInternal [label: "n8n's own storage connection", icon: aws-ec2, color: purple]
yourMac [label: "You on your Mac - scripts/tunnel-rds.sh then scripts/psql.sh (defaults to n8n_workflows, or DB_ROLE=app_ingest)", icon: laptop, color: blue]

// ======================= THE FOUR LOGINS =======================
loginsGroup [label: "LOGINS (Postgres roles) - passwords generated by Secrets Manager, never typed by a human", icon: key, color: red] {
  masterRole [label: "master - everything. Used ONLY by the migrate Lambda, password rotated by RDS every 7 days, fetched fresh each time", icon: shield]
  appIngestRole [label: "app_ingest - for the api + ingest Lambdas", icon: user-check]
  n8nWorkflowsRole [label: "n8n_workflows - the n8n Postgres credential", icon: user-check]
  n8nMetaRole [label: "n8n_meta - owns the n8n database only", icon: user-check]
}

// ======================= THE DATABASE SERVER =======================
rdsGroup [label: "RDS PostgreSQL 16 - private subnets, TLS forced, no internet access", icon: aws-rds, color: blue] {

  appDbGroup [label: "DATABASE app - the business data (17 tables)", icon: database, color: blue] {

    uploadTables [label: "UPLOAD: upload_jobs, users, upload_rejects", icon: table, color: blue]
    progressTables [label: "PROGRESS READ-ONLY for the UI: pipeline_runs, notifications", icon: table, color: purple]
    resultTables [label: "RESULTS: matches, loan_products, criteria_rules, crawl_runs, workflow_errors", icon: table, color: orange]
    matchWorkTables [label: "MATCH WORK: match_classes, match_pairs", icon: table, color: orange]
    crawlConfig [label: "CRAWLER CONFIG: crawl_sources (read-only for n8n, changed only by a migration)", icon: table, color: green]
    dashboardTables [label: "DASHBOARD: user_preferences, accounts, account_profiles", icon: table, color: pink]
    bookkeeping [label: "BOOKKEEPING: schema_migrations", icon: table, color: gray]
  }

  n8nDbGroup [label: "DATABASE n8n - n8n's own notebook (workflows, saved credentials, execution history). n8n creates its own tables here.", icon: database, color: purple] {
    n8nTables [label: "n8n internal tables - not listed, managed by n8n itself", icon: table, color: purple]
  }
}

// ======================= NOTES =======================
noteWhy [label: "NOTE Why two databases: n8n's internals are kept apart from business data, so a rebuilt EC2 keeps its workflows (stored in RDS) and the engine data cannot be damaged by n8n's own migrations.", icon: info, color: yellow]
noteApiDelete [label: "NOTE app_ingest may DELETE only users and account_profiles. Child rows (matches, notifications, preferences) disappear by ON DELETE CASCADE, so no extra DELETE grant is needed.", icon: info, color: yellow]
noteN8nColumns [label: "NOTE n8n may UPDATE only 3 columns of upload_jobs: match_status, match_started_at, matched_at. It can never rewrite a job's counts or status.", icon: info, color: yellow]
noteMaintain [label: "NOTE n8n also has MAINTAIN on match_classes, match_pairs, matches so Workflow B can ANALYZE them between stages (fresh query-planner statistics).", icon: info, color: yellow]
noteNetwork [label: "NOTE RDS accepts port 5432 only from the Lambda security group and the n8n EC2 security group. Your Mac gets in through an SSM tunnel, never the internet.", icon: info, color: yellow]
noteMigrate [label: "NOTE migrate Lambda: applies sql/001 to 006 once each (recorded in schema_migrations), creates the roles, creates the n8n database owned by n8n_meta, then re-applies R__grants.sql on every run.", icon: info, color: yellow]

// ======================= WHO USES WHICH LOGIN =======================
migrateLambda > masterRole: builds everything
apiLambda > appIngestRole: logs in with
ingestLambda > appIngestRole: logs in with
n8nHost > n8nWorkflowsRole: logs in with (Postgres credential in n8n)
n8nInternal > n8nMetaRole: logs in with (DB_POSTGRESDB_USER)
yourMac > n8nWorkflowsRole: psql through SSM tunnel on localhost 15432

// ======================= WHAT EACH LOGIN MAY DO =======================
masterRole > appDbGroup: CREATE tables, roles, grants
masterRole > n8nDbGroup: CREATE DATABASE n8n OWNER n8n_meta

appIngestRole > uploadTables: SELECT INSERT UPDATE (plus DELETE on users)
appIngestRole > progressTables: SELECT only
appIngestRole > resultTables: SELECT only (dashboard + ops view)
appIngestRole > dashboardTables: SELECT INSERT UPDATE (plus DELETE on account_profiles)

n8nWorkflowsRole > uploadTables: SELECT users and upload_jobs, UPDATE 3 progress columns only
n8nWorkflowsRole > progressTables: SELECT INSERT UPDATE
n8nWorkflowsRole > resultTables: SELECT INSERT UPDATE
n8nWorkflowsRole > matchWorkTables: SELECT INSERT UPDATE plus MAINTAIN
n8nWorkflowsRole > crawlConfig: SELECT only

n8nMetaRole > n8nDbGroup: owns it, can reach nothing else

// ======================= WHAT NOBODY CAN DO (dotted) =======================
appIngestRole -- matchWorkTables: NO access
appIngestRole -- n8nDbGroup: NO access
n8nWorkflowsRole -- dashboardTables: NO access
n8nWorkflowsRole -- n8nDbGroup: NO access
n8nMetaRole -- appDbGroup: NO access

// ======================= NOTES ATTACHED =======================
noteWhy -- n8nDbGroup
noteApiDelete -- appIngestRole
noteN8nColumns -- n8nWorkflowsRole
noteMaintain -- matchWorkTables
noteNetwork -- rdsGroup
noteMigrate -- migrateLambda
